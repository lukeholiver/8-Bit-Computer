// load base into %r0 and exponent into %r1
mov $05, %r0
mov $03, %r1
mov $38, %r2

// call exponent function then halt
call exponent
halt

// exponent function label
exponent:

// check for exponent = 0
je %r1, zero

// pre-loop housekeeping
push %r2
mov %r0, %r2
add $FF, %r1

// multiplication loop
loop:
jgt %r1, body
pop %r2
ret

// body of loop
body:
mul %r2, %r0
add $FF, %r1
jmp loop

// exponent = 0
zero:
mov $01, %r0
ret
