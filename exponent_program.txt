// load base into %r0 and exponent into %r1
mov $05, %r0
mov $03, %r1

// call exponent function then halt
call exponent
halt

// exponent function label
exponent:

// pre-loop housekeeping
push %r2
mov %r0, %r2
add $FF, %r1
neg %r1

// multiplication loop
loop:
mul %r2, %r0
add $01, %r1
jlt %r1, loop
pop %r2
ret
